{{- /* Every sentence here is taken from the Radar docs (docs/*.md, radarhq.io/docs)
       and the ingress/HTTPRoute commands from Helm's `helm create` NOTES.txt
       scaffold. Keep it that way: don't add claims the docs don't make.
       Never print a value that can hold a credential. cloud.url is named, not
       printed, because a URL can carry userinfo or a token in its query. */ -}}
{{- $fullName := include "radar.fullname" . -}}
{{- $ns := .Release.Namespace -}}
{{- $port := .Values.service.port -}}
{{- $basePath := include "radar.basePath" . -}}
{{- $authDocs := "https://radarhq.io/docs/configuration/authentication" -}}
{{- $mode := .Values.auth.mode -}}
{{- $routed := or .Values.ingress.enabled .Values.httpRoute.enabled -}}
{{- $exposed := or $routed (ne .Values.service.type "ClusterIP") -}}
{{- $oidcURL := "" -}}
{{- $localHost := "localhost" -}}
{{- $localPort := "9280" -}}
{{- if and (eq $mode "oidc") .Values.auth.oidc.redirectURL (not $routed) -}}
{{- /* Parsed by regex, not urlParse: a malformed URL must not fail the render,
       because Helm's error would print the whole value, userinfo included. */ -}}
{{- $r := .Values.auth.oidc.redirectURL -}}
{{- $scheme := regexFind "^https?://" $r -}}
{{- $host := regexReplaceAll "^.*@" (regexReplaceAll "^https?://([^/?#]*).*$" $r "${1}") "" -}}
{{- if and $scheme (regexMatch "^([A-Za-z0-9.-]+|\\[[0-9A-Fa-f:.]+\\])(:[0-9]+)?$" $host) -}}
{{- $hostOnly := regexReplaceAll ":[0-9]+$" $host "" -}}
{{- /* Only an http loopback redirect can be reached through a port-forward:
       Radar's listener doesn't serve TLS. */ -}}
{{- if and (eq $scheme "http://") (has $hostOnly (list "localhost" "127.0.0.1" "[::1]")) -}}
{{- $localHost = $hostOnly -}}
{{- $localPort = trimPrefix ":" (regexFind ":[0-9]+$" $host) | default "80" -}}
{{- else -}}
{{- $oidcURL = printf "%s%s%s/" $scheme $host $basePath -}}
{{- end -}}
{{- end -}}
{{- end -}}
{{- $ingressAuth := false -}}
{{- range $k, $_ := .Values.ingress.annotations -}}
{{- if regexMatch "auth-(type|url)$" $k }}{{ $ingressAuth = true }}{{ end -}}
{{- end -}}
{{- $proxyAuth := printf "Authentication: proxy (auth.mode=proxy). Users authenticate through the proxy,\nwhich forwards their identity to Radar via HTTP headers. The auth proxy should\nbe the only path to Radar.\nSetup: %s" $authDocs -}}
{{- $proxyWarning := printf "Warning: Radar accepts the %s and %s\nheaders from any client that can reach it. Your ingress must strip them from\nexternal requests to prevent spoofing." .Values.auth.proxy.userHeader .Values.auth.proxy.groupsHeader -}}

{{- $regexURLs := list }}
{{- $hostless := false }}
{{- range $h := .Values.ingress.hosts }}
{{- if not $h.host }}{{ $hostless = true }}{{ end }}
{{- range $h.paths }}
{{- $p := .path | default "/" }}
{{- $scheme := ternary "https" "http" (not (empty $.Values.ingress.tls)) }}
{{- /* A regex path (nginx use-regex + rewrite-target) isn't a URL. */}}
{{- if and (eq (.pathType | default "") "ImplementationSpecific") (regexMatch "[][(){}*+?|^$\\\\]" $p) (eq $basePath "") }}
{{- if $h.host }}
{{- $regexURLs = append $regexURLs (printf "%s://%s  (path %s)" $scheme $h.host $p) }}
{{- end }}
{{- else if $h.host }}
{{- $path := ternary (printf "%s/" $basePath) $p (ne $basePath "") }}
{{- $urls = append $urls (printf "%s://%s%s" $scheme $h.host $path) }}
{{- end }}
{{- end }}
{{- end }}

{{- $listener := ternary (printf "[?(@.name=='%s')]" $gw.sectionName) "[0]" (not (empty $gw.sectionName)) }}
